select 
to_char(o.order_date, 'YYYY-MM') ||'-01' as the_date,
count(o.order_id) as  total_number_of_orders,
round(sum(freight)) as  total_freight
from public.orders o 
where (o.order_date between '1997-01-01' and '1998-12-31') 
group by the_date
having count(o.order_id) >35
order by total_freight desc
;
