SELECT 
  e.first_name || ' ' || e.last_name AS full_name,
  e.title,
  ROUND(SUM(od.unit_price * od.quantity)::numeric, 2) AS total_sales_excl_discount,
  COUNT(DISTINCT od.order_id) AS total_order_unique,
  COUNT(od.order_id) AS total_order,
  ROUND(AVG(od.unit_price * od.quantity)::numeric, 2) AS avg_product_amount,
  ROUND(SUM(od.unit_price * od.quantity)::numeric / NULLIF(COUNT(DISTINCT od.order_id), 0), 2) AS avg_order_amount,
  ROUND(SUM(od.unit_price * od.quantity * od.discount)::numeric, 2) AS total_discount_amount,
  ROUND(SUM(od.unit_price * od.quantity * (1 - od.discount))::numeric, 2) AS total_sales_amount_incl_discount,
  ROUND(
    SUM(od.unit_price * od.quantity * od.discount)::numeric 
    / NULLIF(SUM(od.unit_price * od.quantity)::numeric, 0) * 100, 
    2
  ) AS total_discount_percentage
FROM employees e
JOIN orders o ON e.employee_id = o.employee_id
JOIN order_details od ON o.order_id = od.order_id
GROUP BY e.employee_id
ORDER BY total_sales_amount_incl_discount DESC;
