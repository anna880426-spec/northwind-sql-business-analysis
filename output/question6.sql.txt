select c.category_name , 
case
	when p.unit_price  between 1 and 19 then  '1. Below $20'
	when p.unit_price  between 20 and 50 then '2. $20 - $50'
	else '3. Over $50'
end as price_range,
round(sum(od.quantity * od.unit_price *(1-od.discount ))) as amount,
count(distinct od.order_id ) as number_of_order
from public.products p 
join public.categories c on p.category_id = c.category_id 
join public.order_details od  on p.product_id = od.product_id 
group by c.category_id ,c.category_name, 
case
	when p.unit_price  between 1 and 19 then  '1. Below $20'
	when p.unit_price  between 20 and 50 then '2. $20 - $50'
	else '3. Over $50'
end 
order by c.category_name ,case
	when p.unit_price  between 1 and 19 then  '1. Below $20'
	when p.unit_price  between 20 and 50 then '2. $20 - $50'
	else '3. Over $50'
end    asc
;
